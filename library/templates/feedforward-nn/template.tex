\documentclass[border=6pt]{standalone}

% teXlab library — feed-forward neural net (style benchmark reference)
\usepackage{tikz}
% Edges behind neurons so fills stay clean.
\pgfdeclarelayer{edges}
\pgfsetlayers{edges,main}

\definecolor{tlblue}{HTML}{2E6B8A}
\definecolor{tlteal}{HTML}{3D8B7A}
\definecolor{tlorange}{HTML}{B86B2C}
\definecolor{tlpurple}{HTML}{6E5A7E}
\definecolor{tlgray}{HTML}{3F4A56}

\begin{document}
% ==== parameters (edit these) ============================================
% Neurons per layer, in order. The number of entries IS the number of layers.
\def\layersizes{4,6,6,2}
% One label per layer (same length as \layersizes).
\def\layerlabels{Input,Hidden,Hidden,Output}
\def\layersep{2.5}
\def\neuronsep{1.15}
\def\neuronsize{0.78}
% =========================================================================

\begin{tikzpicture}[
    neuron/.style={
      circle, draw=tlblue!80!black, line width=0.75pt,
      fill=tlblue!12, minimum size=\neuronsize cm, inner sep=0pt,
    },
    edge/.style={draw=tlgray!40, line width=0.35pt},
    layerlabel/.style={font=\sffamily\small, text=tlgray, anchor=base},
  ]

  % Label row sits just below the lowest neuron of the largest layer.
  \pgfmathsetmacro{\maxsize}{0}
  \foreach \s in \layersizes {%
    \pgfmathsetmacro{\maxsize}{max(\maxsize,\s)}\global\let\maxsize\maxsize}
  \pgfmathsetmacro{\labely}{-((\maxsize-1)/2)*\neuronsep-0.7}

  % Place layers left→right, vertically centred on y=0; fully connect to previous.
  \foreach \size [count=\layer, remember=\size as \prevsize (initially 0)] in \layersizes {
    \pgfmathsetmacro{\x}{\layer*\layersep}
    \foreach \n in {1,...,\size} {
      \pgfmathsetmacro{\y}{(\n-(\size+1)/2)*\neuronsep}
      \node[neuron] (L\layer-\n) at (\x,\y) {};
    }
    \ifnum\layer>1\relax
      \pgfmathtruncatemacro{\prevlayer}{\layer-1}
      \begin{pgfonlayer}{edges}
        \foreach \p in {1,...,\prevsize} {
          \foreach \q in {1,...,\size} {
            \draw[edge] (L\prevlayer-\p) -- (L\layer-\q);
          }
        }
      \end{pgfonlayer}
    \fi
  }

  \foreach \lab [count=\layer] in \layerlabels {
    \pgfmathsetmacro{\x}{\layer*\layersep}
    \node[layerlabel] at (\x,\labely) {\lab};
  }

\end{tikzpicture}
\end{document}
